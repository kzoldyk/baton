Baton scans your machine's system PATH to detect which agent CLIs you have installed.

const installed = await Promise.all(
  AGENT_CONFIGS.map(async (agent) => {
    const found = await this.checkBinary(agent.bin);
    return found ? agent.id : null;
  })
